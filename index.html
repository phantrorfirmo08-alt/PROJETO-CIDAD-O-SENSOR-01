function FieldReportModule({ onBackToChat, showToast }) { // Recupera rascunho salvo anteriormente se houver const [setor, setSetor] = React.useState(() => localStorage.getItem("sarah_vistoria_setor") || "🛡️ Defesa Civil (Vistoria de Risco / Ocorrência)"); const [agente, setAgente] = React.useState(() => localStorage.getItem("sarah_vistoria_agente") || ""); const [localizacao, setLocalizacao] = React.useState(() => localStorage.getItem("sarah_vistoria_local") || ""); const [descricao, setDescricao] = React.useState(() => localStorage.getItem("sarah_vistoria_desc") || ""); const [gpsCoords, setGpsCoords] = React.useState({ lt: '', ln: '', lk: '' }); const [gpsStatus, setGpsStatus] = React.useState("📍 GPS em espera"); const [fotos, setFotos] = React.useState([ { id: 1, status: "vazia", file: null, preview: null }, { id: 2, status: "vazia", file: null, preview: null }, { id: 3, status: "vazia", file: null, preview: null }, { id: 4, status: "vazia", file: null, preview: null } ]); // Salva rascunho automaticamente para evitar perda de dados em caso de reset React.useEffect(() => { try { localStorage.setItem("sarah_vistoria_setor", setor); localStorage.setItem("sarah_vistoria_agente", agente); localStorage.setItem("sarah_vistoria_local", localizacao); localStorage.setItem("sarah_vistoria_desc", descricao); } catch(e) {} }, [setor, agente, localizacao, descricao]); const capturarGps = () => { if (!navigator.geolocation) { showToast("⚠️ Geolocalização não suportada."); return; } showToast("🛰 Acionando GPS de alta precisão..."); navigator.geolocation.getCurrentPosition( (p) => { const lt = p.coords.latitude.toFixed(6); const ln = p.coords.longitude.toFixed(6); const lk = `https://maps.google.com/?q=${lt},${ln}`; setGpsCoords({ lt, ln, lk }); setLocalizacao(`${lt},${ln}`); setGpsStatus("✅ GPS Fixado com Sucesso"); showToast(`📍 Coordenadas obtidas: ${lt},${ln}`); }, () => { showToast("⚠ Falha ao obter sinal GPS. Verifique permissões."); setGpsStatus("⚠ Erro de Sinal GPS"); }, { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 } ); }; const processarFotoVistoria = (file) => { return new Promise((resolve, reject) => { const reader = new FileReader(); reader.readAsDataURL(file); reader.onload = (event) => { const img = new Image(); img.src = event.target.result; img.onload = () => { const canvas = document.createElement('canvas'); let width = img.width; let height = img.height; const MAX_DIMENSION = 1024; // Reduzido para 1024 para garantir estabilidade de memória em celulares if (width > height) { if (width > MAX_DIMENSION) { height = Math.round((height * MAX_DIMENSION) / width); width = MAX_DIMENSION; } } else { if (height > MAX_DIMENSION) { width = Math.round((width * MAX_DIMENSION) / height); height = MAX_DIMENSION; } } canvas.width = width; canvas.height = height; const ctx = canvas.getContext('2d'); ctx.imageSmoothingEnabled = true; ctx.imageSmoothingQuality = 'medium'; ctx.drawImage(img, 0, 0, width, height); canvas.toBlob((blob) => { if (!blob) { reject(new Error("Falha ao gerar blob da imagem comprimida")); return; } const compressedFile = new File([blob], file.name.replace(/\.[^/.]+$/, "") + ".jpg", { type: 'image/jpeg', lastModified: Date.now() }); const previewUrl = URL.createObjectURL(compressedFile); resolve({ compressedFile, previewUrl }); }, 'image/jpeg', 0.70); // Qualidade 70% otimizada para envio rápido e sem travar }; img.onerror = reject; }; reader.onerror = reject; }); }; const handleFotoChange = async (i, e) => { const file = e.target.files[0]; if (file) { showToast(`🔄 Otimizando evidência ${i + 1}...`); try { const { compressedFile, previewUrl } = await processarFotoVistoria(file); setFotos(prev => { const u = [...prev]; // Libera URL anterior se existir para evitar vazamento de memória if (u[i].preview) { URL.revokeObjectURL(u[i].preview); } u[i] = { id: u[i].id, status: "✅ OK", file: compressedFile, preview: previewUrl }; return u; }); showToast(`📸 Evidência ${i + 1} otimizada com sucesso!`); } catch (err) { console.error(err); showToast(`❌ Erro ao processar foto ${i + 1}. Tente outra imagem.`); } } }; const montarTexto = () => { const fCount = fotos.filter(f => f.status === "✅ OK").length; return `📊 *RELATÓRIO TÉCNICO DE CAMPO - SARAH IA*\n` + `🏛 *Setor:* ${setor}\n` + `👤 *Agente:* ${agente || 'Não informado'}\n` + `📍 *Local:* ${localizacao || 'Não informado'}\n` + (gpsCoords.lk ? `🔗 *Geolocalização:* ${gpsCoords.lk}\n` : ``) + `📸 *Evidências fotográficas:* ${fCount}/4\n` + `📝 *Relato Operacional:*\n${descricao || 'Sem descrição informada'}`; }; const copiarLaudo = () => { const texto = montarTexto(); navigator.clipboard.writeText(texto).then(() => { showToast("📋 Relatório copiado para a área de transferência!"); }).catch(() => { showToast("❌ Falha ao copiar relatório."); }); }; const enviarZapComFotos = async () => { const texto = montarTexto(); const validFiles = fotos.filter(f => f.file instanceof File).map(f => f.file); // Tenta compartilhamento nativo seguro com tratamento try/catch para evitar crash if (navigator.share && validFiles.length > 0 && navigator.canShare && navigator.canShare({ files: validFiles })) { try { await navigator.share({ title: 'Relatório Técnico - SARAH IA', text: texto, files: validFiles }); showToast("✅ Relatório e evidências compartilhados com sucesso!"); return; } catch (err) { // Se o usuário cancelar ou o navegador bloquear o envio de arquivos, faz fallback automático para texto sem quebrar o app if (err.name === 'AbortError') { showToast("ℹ️ Compartilhamento cancelado pelo usuário."); return; } console.warn("Falha no nativa share com arquivos, alternando para WhatsApp Web...", err); } } // Fallback seguro: abre o WhatsApp com o texto formatado e copia automaticamente o laudo navigator.clipboard.writeText(texto).catch(() => {}); window.open(`https://api.whatsapp.com/send?text=${encodeURIComponent(texto)}`, '_blank'); showToast("💬 WhatsApp aberto! (Texto copiado para colar caso necessário)"); }; // ... restante do JSX do módulo ...
